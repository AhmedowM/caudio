cmake_minimum_required(VERSION 3.28)
project(caudio_module_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES ON)
find_package(caudio CONFIG REQUIRED)

add_executable(module_media media.cpp)
target_link_libraries(module_media PRIVATE caudio::media)
add_executable(module_library library.cpp)
target_link_libraries(module_library PRIVATE caudio::db caudio::engine)
