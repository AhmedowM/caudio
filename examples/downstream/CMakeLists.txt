cmake_minimum_required(VERSION 3.28)
project(caudio_downstream_examples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(caudio CONFIG REQUIRED)

function(add_caudio_example name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE ${ARGN})
endfunction()

add_caudio_example(example_types types.cpp caudio::types)
add_caudio_example(example_contracts contracts.cpp caudio::contracts)
add_caudio_example(example_utils utils.cpp caudio::utils)
add_caudio_example(example_config config.cpp caudio::config)
add_caudio_example(example_media media.cpp caudio::media)
add_caudio_example(example_player player.cpp caudio::player)
add_caudio_example(example_database database.cpp caudio::db)
add_caudio_example(example_engine engine.cpp caudio::engine caudio::db caudio::player)
add_caudio_example(example_ipc ipc.cpp caudio::ipc)
add_caudio_example(example_client client.cpp caudio::client)
add_caudio_example(example_service service.cpp caudio::service)
add_caudio_example(example_app app.cpp caudio::app)
add_caudio_example(example_library library.cpp caudio::engine caudio::db caudio::utils)
add_caudio_example(example_fingerprint fingerprint.cpp caudio::media caudio::utils)
add_caudio_example(example_config_backup config_backup.cpp caudio::config caudio::utils)
add_caudio_example(example_mini_player mini_player.cpp caudio::player caudio::utils)
